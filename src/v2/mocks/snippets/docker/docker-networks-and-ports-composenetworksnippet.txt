
# docker-compose.yml - автоматические сети
version: '3.8'

services:
  web:
    image: nginx
    ports:
      - "80:80"        # host:container
      - "443:443"
    networks:
      - frontend

  api:
    build: .
    expose:
      - "3000"         # Только внутри compose сети
    networks:
      - frontend
      - backend

  db:
    image: postgres
    # НЕТ ports - недоступна снаружи
    # ЕСТЬ expose внутри образа
    networks:
      - backend
    environment:
      POSTGRES_DB: myapp

networks:
  frontend:
    driver: bridge
  backend:
    driver: bridge
