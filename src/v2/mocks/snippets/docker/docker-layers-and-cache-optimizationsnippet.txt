
# Продвинутые техники оптимизации слоев

# 1. Группировка RUN команд для уменьшения слоев
RUN apt-get update &&     apt-get install -y         curl         git         python3 &&     apt-get clean &&     rm -rf /var/lib/apt/lists/*

# 2. Multi-stage builds для минимизации финального образа
FROM node:18-alpine AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build

FROM nginx:alpine AS production
COPY --from=builder /app/dist /usr/share/nginx/html
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]

# 3. Использование .dockerignore для стабильности кэша
# В .dockerignore:
# node_modules/
# .git/
# *.log
# tests/
