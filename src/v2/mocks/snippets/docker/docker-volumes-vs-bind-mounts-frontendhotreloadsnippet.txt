
# docker-compose.frontend.yml - React/Vue hot reload
version: '3.8'

services:
  frontend:
    build:
      context: .
      dockerfile: Dockerfile.dev
    ports:
      - "3000:3000"
    environment:
      - CHOKIDAR_USEPOLLING=true
      - WATCHPACK_POLLING=true
      - WDS_SOCKET_HOST=localhost
      - WDS_SOCKET_PORT=3000
    volumes:
      # Исходный код
      - .:/app
      # Исключаем node_modules
      - /app/node_modules
      # Исключаем build
      - /app/build
    stdin_open: true
    tty: true
