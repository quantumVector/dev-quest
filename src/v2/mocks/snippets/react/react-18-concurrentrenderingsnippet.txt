
// React 17 - синхронный рендеринг (blocking)
function LegacyApp() {
  const [count, setCount] = useState(0);

  // Все обновления блокируют UI
  const handleClick = () => {
    setCount(count + 1); // Блокирует до завершения
  };

  return <button onClick={handleClick}>{count}</button>;
}
