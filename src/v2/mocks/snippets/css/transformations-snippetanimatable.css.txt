
/* ✅ Анимируемые свойства — плавный переход */
.box {
  transform: scale(1);
  transform-origin: top left;
  perspective: 1000px;
  perspective-origin: 50% 50%;
  transition: transform 0.3s ease, perspective 0.3s ease;
}
.box:hover {
  transform: scale(1.2) rotate(10deg);
  perspective: 600px;
}
